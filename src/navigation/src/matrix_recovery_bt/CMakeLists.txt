cmake_minimum_required(VERSION 3.16)
project(matrix_recovery_bt)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
set(dependencies rclcpp nav2_util nav2_costmap_2d behaviortree_cpp nav_msgs nav2_msgs
  geometry_msgs sensor_msgs std_msgs action_msgs tf2_ros tf2_geometry_msgs)
foreach(dep IN LISTS dependencies)
  find_package(${dep} REQUIRED)
endforeach()
rosidl_generate_interfaces(${PROJECT_NAME} "srv/RecoveryHandoff.srv")
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
add_executable(zero_handoff_gate src/zero_handoff_gate.cpp)
target_include_directories(zero_handoff_gate PRIVATE include)
ament_target_dependencies(zero_handoff_gate rclcpp geometry_msgs std_msgs)
target_link_libraries(zero_handoff_gate "${typesupport}")
add_library(matrix_recovery_nodes SHARED src/recovery_nodes.cpp)
target_compile_definitions(matrix_recovery_nodes PRIVATE BT_PLUGIN_EXPORT)
target_include_directories(matrix_recovery_nodes PRIVATE include ../robot_feasible_navigation/include)
ament_target_dependencies(matrix_recovery_nodes ${dependencies})
target_link_libraries(matrix_recovery_nodes "${typesupport}")
install(TARGETS matrix_recovery_nodes LIBRARY DESTINATION lib)
install(TARGETS zero_handoff_gate RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_stop_barrier test/test_stop_barrier.cpp)
  target_include_directories(test_stop_barrier PRIVATE include)
  find_package(rclcpp_action REQUIRED)
  ament_add_gtest(test_native_handoff test/test_native_handoff.cpp TIMEOUT 120)
  target_include_directories(test_native_handoff PRIVATE include)
  ament_target_dependencies(test_native_handoff ${dependencies} rclcpp_action)
  target_link_libraries(test_native_handoff "${typesupport}")
  target_compile_definitions(test_native_handoff PRIVATE
    GATE_PATH="${CMAKE_INSTALL_PREFIX}/lib/matrix_recovery_bt/zero_handoff_gate"
    PLUGIN_PATH="${CMAKE_INSTALL_PREFIX}/lib/libmatrix_recovery_nodes.so")
  set_tests_properties(test_native_handoff PROPERTIES ENVIRONMENT
    "ROS_DOMAIN_ID=190;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST;ROS_LOG_DIR=${CMAKE_CURRENT_BINARY_DIR}/ros_logs;AMENT_PREFIX_PATH=${CMAKE_INSTALL_PREFIX}:/opt/ros/jazzy;LD_LIBRARY_PATH=${CMAKE_INSTALL_PREFIX}/lib:/opt/ros/jazzy/lib:/opt/ros/jazzy/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
endif()
ament_export_dependencies(rosidl_default_runtime)
ament_package()
