cmake_minimum_required(VERSION 3.16)
project(matrix_waypoint_follower)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
set(dependencies rclcpp rclcpp_action rclcpp_lifecycle rclcpp_components nav_msgs
  nav2_msgs nav2_util tf2_ros nav2_core pluginlib robot_localization geographic_msgs)
foreach(dep IN LISTS dependencies)
  find_package(${dep} REQUIRED)
endforeach()
# Unique package/SONAME: never replace the installed or legacy follower core.
add_library(matrix_waypoint_follower_core SHARED src/waypoint_follower.cpp)
target_include_directories(matrix_waypoint_follower_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(matrix_waypoint_follower_core ${dependencies})
add_executable(waypoint_follower src/main.cpp)
ament_target_dependencies(waypoint_follower ${dependencies})
target_link_libraries(waypoint_follower matrix_waypoint_follower_core)
install(TARGETS matrix_waypoint_follower_core LIBRARY DESTINATION lib)
install(TARGETS waypoint_follower RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(FILES UPSTREAM.json callback_nonblocking.patch LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(callback_probe test/callback_probe.cpp)
  ament_target_dependencies(callback_probe ${dependencies})
  target_link_libraries(callback_probe matrix_waypoint_follower_core)
  foreach(mode normal inverted reject)
    add_test(NAME callback_${mode} COMMAND callback_probe ${mode})
    set_tests_properties(callback_${mode} PROPERTIES TIMEOUT 12
      PASS_REGULAR_EXPRESSION "GUARDED_ASSERTIONS_PASSED"
      ENVIRONMENT "ROS_DOMAIN_ID=190;ROS_AUTOMATIC_DISCOVERY_RANGE=LOCALHOST;ROS_LOG_DIR=${CMAKE_CURRENT_BINARY_DIR}/ros_logs;AMENT_PREFIX_PATH=/opt/ros/jazzy;LD_LIBRARY_PATH=${CMAKE_CURRENT_BINARY_DIR}:/opt/ros/jazzy/lib:/opt/ros/jazzy/lib/${CMAKE_LIBRARY_ARCHITECTURE}")
  endforeach()
endif()
ament_export_include_directories(include)
ament_export_libraries(matrix_waypoint_follower_core)
ament_export_dependencies(${dependencies})
ament_package()
