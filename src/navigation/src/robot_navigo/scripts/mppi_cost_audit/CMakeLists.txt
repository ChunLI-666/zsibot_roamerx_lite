cmake_minimum_required(VERSION 3.16)
project(mppi_cost_audit)
find_package(ament_cmake REQUIRED)
find_package(navigo_mppi_controller REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(xtensor REQUIRED)
find_package(xsimd REQUIRED)
add_executable(cost_probe cost_probe.cpp)
target_compile_features(cost_probe PRIVATE cxx_std_17)
target_compile_options(cost_probe PRIVATE -O2 -mno-avx512f -msse4.2 -mavx2 -mfma)
target_compile_definitions(cost_probe PRIVATE XTENSOR_ENABLE_XSIMD XTENSOR_USE_XSIMD)
ament_target_dependencies(cost_probe navigo_mppi_controller)
target_link_libraries(cost_probe yaml-cpp xtensor xtensor::optimize xtensor::use_xsimd)
